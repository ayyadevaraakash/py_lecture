def is_prime(n):
    if n < 2:
        return False
    den = 2
    while den*den <= n:
        if n%den==0:
            return False
        den = den + 1
    return True

n = int(input())
primes = []
for i in range(2,n//2+1):
    if is_prime(i):
        primes.append(i)

# print(primes)
ans = []
sqrt = 1
for num in primes:
    cnt = 0
    while n%num==0:
        n=n//num
        ans.append(num)
        cnt+=1
    sqrt = sqrt * (num**(cnt//2))

if n!=1:
    ans.append(n)

print(ans)
print(sqrt)
