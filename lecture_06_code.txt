import numpy as np

n = int(input("Rows in A : "))
m = int(input("Cols in A : "))

A = np.zeros((n,m))

for i in range(n):
	for j in range(m):
		val = int(input(f"A[{i}][{j}] = "))
		A[i][j] = val

p = int(input("Rows in B : "))
q = int(input("Cols in B : "))

B = np.zeros((p,q))

for i in range(p):
	for j in range(q):
		val = int(input(f"B[{i}][{j}] = "))
		B[i][j] = val

if m==p:
	print(A @ B)
else:
	print("Matrix are incompatible to each other")
